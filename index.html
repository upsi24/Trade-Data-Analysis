//@version=6 strategy( title = "Fast XAUUSD Scalp Pro v6 [M1/M5/M15]", shorttitle = "XAU Scalp Pro v6", overlay = true, initial_capital = 10000, default_qty_type = strategy.fixed, default_qty_value = 1, currency = currency.USD, commission_type = strategy.commission.cash_per_contract, commission_value = 0.0, slippage = 1, pyramiding = 0, process_orders_on_close = true, max_labels_count = 150, max_lines_count = 60 ) // ============================================================================ // 1. INPUTS // ============================================================================ grpSpeed = "1. Speed & Signal Sensitivity" speedMode = input.string("Fast Scalp", "Scalp Speed Mode", options = ["Fast Scalp", "Balanced", "High Accuracy"], group = grpSpeed) fastLen = input.int(9, "Fast EMA Length", minval = 2, group = grpSpeed) slowLen = input.int(21, "Slow EMA Length", minval = 3, group = grpSpeed) stAtrLen = input.int(10, "Supertrend ATR Length", minval = 2, group = grpSpeed) allowPullback = input.bool(true, "Enable Trend Pullback Entries", group = grpSpeed) cooldownBars = input.int(3, "Min Bars Between Same-Direction Signals", minval = 1, group = grpSpeed) grpFilter = "2. Multi-Timeframe (M5 / M15) & Clarity Filters" useM5Filter = input.bool(true, "Use M5 Confirmation Weight", group = grpFilter) useM15Filter = input.bool(true, "Use M15 Trend Bias Filter", group = grpFilter) useAdxFilter = input.bool(true, "ADX Chop Filter (Show WAIT in Sideways)", group = grpFilter) adxLen = input.int(14, "ADX Length", minval = 5, group = grpFilter) rsiLen = input.int(14, "RSI Length", minval = 5, group = grpFilter) minProbCustom = input.int(60, "Min Entry Probability % (Balanced Reference)", minval = 52, maxval = 85, group = grpFilter) grpRisk = "3. Stop Loss & Collect Profit Management" slAtrMult = input.float(1.3, "Stop Loss (ATR Multiplier)", minval = 0.5, step = 0.1, group = grpRisk) tp1RR = input.float(1.5, "TP1 Collect Profit (Risk : Reward)", minval = 0.5, step = 0.1, group = grpRisk) tp2RR = input.float(2.5, "TP2 Runner Target (Risk : Reward)", minval = 1.0, step = 0.1, group = grpRisk) tp1ClosePct = input.int(50, "TP1 Collect Profit Size (%)", minval = 10, maxval = 100, step = 10, group = grpRisk) moveBeOnTp1 = input.bool(true, "Auto Move Stop Loss to Break-Even at TP1", group = grpRisk) closeOnOpp = input.bool(false, "Exit Early on Opposite Strong Signal", group = grpRisk) grpVisual = "4. Chart Visuals & Dashboard" showRibbon = input.bool(true, "Show EMA 9/21 Scalp Ribbon", group = grpVisual) showRiskLines = input.bool(true, "Plot Entry / SL / TP1 / TP2 Lines", group = grpVisual) showDashboard = input.bool(true, "Show Live Signal & Probability HUD", group = grpVisual) // Mode-adjusted thresholds (Fast Scalp triggers quickest; High Accuracy filters hardest) stFactor = speedMode == "Fast Scalp" ? 1.4 : speedMode == "Balanced" ? 1.7 : 2.1 adxMin = speedMode == "Fast Scalp" ? 14.0 : speedMode == "Balanced" ? 18.0 : 22.0 minProb = speedMode == "Fast Scalp" ? math.max(56, minProbCustom - 4) : speedMode == "Balanced" ? minProbCustom : math.min(80, minProbCustom + 6) rsiHi = speedMode == "Fast Scalp" ? 78.0 : 74.0 rsiLo = speedMode == "Fast Scalp" ? 22.0 : 26.0 // ============================================================================ // 2. CORE & MULTI-TIMEFRAME (M1 / M5 / M15) CALCULATIONS (NON-REPAINTING) // ============================================================================ emaFast = ta.ema(close, fastLen) emaSlow = ta.ema(close, slowLen) emaTrend = ta.ema(close, 50) [stLine, stDir] = ta.supertrend(stFactor, stAtrLen) [_, _, adx] = ta.dmi(adxLen, adxLen) rsi = ta.rsi(close, rsiLen) atr = math.max(ta.atr(14), syminfo.mintick * 10) // Reusable timeframe scorer (-1.0 to +1.0) f_tfScore() => _ef = ta.ema(close, fastLen) _es = ta.ema(close, slowLen) _rsi = ta.rsi(close, rsiLen) _atr = math.max(ta.atr(14), syminfo.mintick * 10) _sep = math.min(math.abs(_ef - _es) / _atr, 1.5) _dir = _ef > _es ? 1.0 : _ef < _es ? -1.0 : 0.0 _mom = _rsi > 53 ? 0.35 : _rsi < 47 ? -0.35 : 0.0 _prc = close > _ef ? 0.20 : close < _ef ? -0.20 : 0.0 math.max(-1.0, math.min(1.0, _dir * (0.30 + _sep * 0.25) + _mom + _prc)) // Fetch confirmed (closed-bar) M5 and M15 scores with zero lookahead / repainting m1ScoreRaw = f_tfScore() m5ScoreRaw = request.security(syminfo.tickerid, "5", f_tfScore()[1], gaps = barmerge.gaps_off, lookahead = barmerge.lookahead_on) m15ScoreRaw = request.security(syminfo.tickerid, "15", f_tfScore()[1], gaps = barmerge.gaps_off, lookahead = barmerge.lookahead_on) m1Score = nz(m1ScoreRaw, 0.0) m5Score = nz(m5ScoreRaw, 0.0) m15Score = nz(m15ScoreRaw, 0.0) // Directional labels per timeframe f_biasText(float sc) => sc >= 0.20 ? "BULL" : sc <= -0.20 ? "BEAR" : "FLAT" m1Bias = f_biasText(m1Score) m5Bias = f_biasText(m5Score) m15Bias = f_biasText(m15Score) // ============================================================================ // 3. PROBABILITY ENGINE & MARKET CLARITY ("WAIT") LOGIC // ============================================================================ // Candle body momentum contribution barRange = math.max(high - low, syminfo.mintick) bodyComp = math.max(-0.20, math.min(0.20, ((close - open) / barRange) * 0.20)) stComp = stDir < 0 ? 0.25 : -0.25 wChart = 0.40 wM5 = useM5Filter ? 0.25 : 0.0 wM15 = useM15Filter ? 0.15 : 0.0 wSt = 0.20 wTotal = wChart + wM5 + wM15 + wSt composite = (m1Score * wChart + m5Score * wM5 + m15Score * wM15 + (stComp + bodyComp) * wSt) / wTotal buyProb = math.round(math.max(5.0, math.min(95.0, 50.0 + composite * 46.0))) sellProb = 100 - buyProb nextDir = buyProb >= 50 ? "BUY" : "SELL" topProb = math.max(buyProb, sellProb) // Clarity guardrails isChoppy = useAdxFilter and adx < adxMin m5ConflictB = useM5Filter and m5Bias == "BEAR" m5ConflictS = useM5Filter and m5Bias == "BULL" m15HardBlockB = useM15Filter and m15Bias == "BEAR" and buyProb < 72 m15HardBlockS = useM15Filter and m15Bias == "BULL" and sellProb < 72 rsiOverbought = rsi > rsiHi rsiOversold = rsi < rsiLo // Fast scalping triggers stFlipBull = stDir < 0 and nz(stDir[1], 1) > 0 stFlipBear = stDir > 0 and nz(stDir[1], -1) < 0 emaCrossBull = ta.crossover(emaFast, emaSlow) and stDir < 0 emaCrossBear = ta.crossunder(emaFast, emaSlow) and stDir > 0 pullbackBull = allowPullback and stDir < 0 and emaFast > emaSlow and low <= emaFast and close > emaFast and close > open pullbackBear = allowPullback and stDir > 0 and emaFast < emaSlow and high >= emaFast and close < emaFast and close < open momBurstBull = speedMode == "Fast Scalp" and stDir < 0 and emaFast > emaSlow and close > high[1] and close > open and buyProb >= minProb + 4 momBurstBear = speedMode == "Fast Scalp" and stDir > 0 and emaFast < emaSlow and close < low[1] and close < open and sellProb >= minProb + 4 rawBuyTrig = stFlipBull or emaCrossBull or pullbackBull or momBurstBull rawSellTrig = stFlipBear or emaCrossBear or pullbackBear or momBurstBear buyClear = not isChoppy and not m5ConflictB and not m15HardBlockB and not rsiOverbought and buyProb >= minProb sellClear = not isChoppy and not m5ConflictS and not m15HardBlockS and not rsiOversold and sellProb >= minProb // Cooldown to avoid duplicate spam var int lastSigDir = 0 var int lastSigBar = -1000 canBuy = rawBuyTrig and buyClear and barstate.isconfirmed and (lastSigDir != 1 or (bar_index - lastSigBar) >= cooldownBars) canSell = rawSellTrig and sellClear and barstate.isconfirmed and (lastSigDir != -1 or (bar_index - lastSigBar) >= cooldownBars) if canBuy lastSigDir := 1 lastSigBar := bar_index else if canSell lastSigDir := -1 lastSigBar := bar_index // Explain why the market is WAIT if no active entry waitReason = isChoppy ? "Choppy Market (ADX " + str.tostring(adx, "#.0") + " < " + str.tostring(adxMin, "#.0") + ")" : (nextDir == "BUY" and m5ConflictB) ? "M5 Bearish Conflict" : (nextDir == "SELL" and m5ConflictS) ? "M5 Bullish Conflict" : (nextDir == "BUY" and m15HardBlockB) ? "Counter-M15 Bear Trend" : (nextDir == "SELL" and m15HardBlockS) ? "Counter-M15 Bull Trend" : (nextDir == "BUY" and rsiOverbought) ? "RSI Overbought (" + str.tostring(rsi, "#.0") + ")" : (nextDir == "SELL" and rsiOversold) ? "RSI Oversold (" + str.tostring(rsi, "#.0") + ")" : topProb < minProb ? "Low Probability (" + str.tostring(topProb) + "% < " + str.tostring(minProb) + "%)" : "Waiting for Pullback / Trigger" // ============================================================================ // 4. STRATEGY EXECUTION: ENTRY, STOP LOSS, COLLECT PROFIT (TP1 / TP2) & BE // ============================================================================ swingLow5 = ta.lowest(low, 5) swingHigh5 = ta.highest(high, 5) var float trEntry = na var float trSL = na var float trTP1 = na var float trTP2 = na var bool tp1Done = false var int trSide = 0 // Optional early reversal exit if closeOnOpp and strategy.position_size > 0 and canSell strategy.close("Long", comment = "Opp Signal Exit") if closeOnOpp and strategy.position_size < 0 and canBuy strategy.close("Short", comment = "Opp Signal Exit") // Open new Long if canBuy and strategy.position_size == 0 rawRisk = math.max(atr * slAtrMult, close - swingLow5 + atr * 0.15) riskDist = math.min(math.max(rawRisk, atr * 0.8), atr * 2.4) trEntry := close trSL := close - riskDist trTP1 := close + riskDist * tp1RR trTP2 := close + riskDist * tp2RR tp1Done := false trSide := 1 strategy.entry("Long", strategy.long, comment = "BUY " + str.tostring(buyProb) + "%") alert("XAUUSD BUY SCALP (" + str.tostring(buyProb) + "%) | Entry: " + str.tostring(trEntry, format.mintick) + " | SL: " + str.tostring(trSL, format.mintick) + " | TP1: " + str.tostring(trTP1, format.mintick) + " | TP2: " + str.tostring(trTP2, format.mintick), alert.freq_once_per_bar_close) // Open new Short if canSell and strategy.position_size == 0 rawRisk = math.max(atr * slAtrMult, swingHigh5 - close + atr * 0.15) riskDist = math.min(math.max(rawRisk, atr * 0.8), atr * 2.4) trEntry := close trSL := close + riskDist trTP1 := close - riskDist * tp1RR trTP2 := close - riskDist * tp2RR tp1Done := false trSide := -1 strategy.entry("Short", strategy.short, comment = "SELL " + str.tostring(sellProb) + "%") alert("XAUUSD SELL SCALP (" + str.tostring(sellProb) + "%) | Entry: " + str.tostring(trEntry, format.mintick) + " | SL: " + str.tostring(trSL, format.mintick) + " | TP1: " + str.tostring(trTP1, format.mintick) + " | TP2: " + str.tostring(trTP2, format.mintick), alert.freq_once_per_bar_close) // Manage Active Long (TP1 partial collect + auto Break-Even + TP2 runner) if strategy.position_size > 0 if not tp1Done and high >= trTP1 tp1Done := true if moveBeOnTp1 trSL := math.max(trSL, strategy.position_avg_price) alert("XAUUSD LONG TP1 COLLECT PROFIT HIT @ " + str.tostring(trTP1, format.mintick) + (moveBeOnTp1 ? " | SL moved to Break-Even" : ""), alert.freq_once_per_bar) strategy.exit("Long TP1", from_entry = "Long", qty_percent = tp1ClosePct, limit = trTP1, stop = trSL, comment_profit = "Collect TP1", comment_loss = "Stop Loss") strategy.exit("Long TP2", from_entry = "Long", qty_percent = 100, limit = trTP2, stop = trSL, comment_profit = "Collect TP2", comment_loss = tp1Done and moveBeOnTp1 ? "BE Stop" : "Stop Loss") // Manage Active Short (TP1 partial collect + auto Break-Even + TP2 runner) if strategy.position_size < 0 if not tp1Done and low <= trTP1 tp1Done := true if moveBeOnTp1 trSL := math.min(trSL, strategy.position_avg_price) alert("XAUUSD SHORT TP1 COLLECT PROFIT HIT @ " + str.tostring(trTP1, format.mintick) + (moveBeOnTp1 ? " | SL moved to Break-Even" : ""), alert.freq_once_per_bar) strategy.exit("Short TP1", from_entry = "Short", qty_percent = tp1ClosePct, limit = trTP1, stop = trSL, comment_profit = "Collect TP1", comment_loss = "Stop Loss") strategy.exit("Short TP2", from_entry = "Short", qty_percent = 100, limit = trTP2, stop = trSL, comment_profit = "Collect TP2", comment_loss = tp1Done and moveBeOnTp1 ? "BE Stop" : "Stop Loss") // Reset tracker when flat if strategy.position_size == 0 and not canBuy and not canSell trSide := 0 // ============================================================================ // 5. CHART VISUALS: RIBBON, SIGNALS & DYNAMIC SL/TP LINES // ============================================================================ ribCol = emaFast >= emaSlow ? #00c896 : #ff4d6d pFast = plot(showRibbon ? emaFast : na, "EMA 9", color = color.new(ribCol, 20), linewidth = 1) pSlow = plot(showRibbon ? emaSlow : na, "EMA 21", color = color.new(ribCol, 20), linewidth = 1) fill(pFast, pSlow, color = showRibbon ? color.new(ribCol, 84) : na, title = "Scalp Ribbon") plot(emaTrend, "EMA 50 Trend", color = color.new(#eaca82, 45), linewidth = 2) plotshape(canBuy, title = "BUY Signal", text = "BUY", style = shape.labelup, location = location.belowbar, color = #00c896, textcolor = color.white, size = size.small) plotshape(canSell, title = "SELL Signal", text = "SELL", style = shape.labeldown, location = location.abovebar, color = #ff4d6d, textcolor = color.white, size = size.small) var line lnEntry = na var line lnSL = na var line lnTP1 = na var line lnTP2 = na var label lbEntry = na var label lbSL = na var label lbTP1 = na var label lbTP2 = na if showRiskLines and (canBuy or canSell or strategy.position_size != 0) if not na(lnEntry) line.delete(lnEntry) line.delete(lnSL) line.delete(lnTP1) line.delete(lnTP2) label.delete(lbEntry) label.delete(lbSL) label.delete(lbTP1) label.delete(lbTP2) x1 = bar_index x2 = bar_index + 12 lnEntry := line.new(x1, trEntry, x2, trEntry, color = #eaca82, style = line.style_dotted, width = 1) lnSL := line.new(x1, trSL, x2, trSL, color = #ff4d6d, style = line.style_dashed, width = 1) lnTP1 := line.new(x1, trTP1, x2, trTP1, color = #00c896, style = line.style_dashed, width = 1) lnTP2 := line.new(x1, trTP2, x2, trTP2, color = #38bdf8, style = line.style_dashed, width = 1) lbEntry := label.new(x2, trEntry, "ENTRY " + str.tostring(trEntry, format.mintick), style = label.style_label_left, color = color.new(#eaca82, 15), textcolor = color.black, size = size.tiny) lbSL := label.new(x2, trSL, (tp1Done and moveBeOnTp1 ? "BE " : "SL ") + str.tostring(trSL, format.mintick), style = label.style_label_left, color = color.new(#ff4d6d, 15), textcolor = color.white, size = size.tiny) lbTP1 := label.new(x2, trTP1, "TP1 " + str.tostring(trTP1, format.mintick), style = label.style_label_left, color = color.new(#00c896, 15), textcolor = color.white, size = size.tiny) lbTP2 := label.new(x2, trTP2, "TP2 " + str.tostring(trTP2, format.mintick), style = label.style_label_left, color = color.new(#0284c7, 15), textcolor = color.white, size = size.tiny) // ============================================================================ // 6. REAL-TIME HUD DASHBOARD (SIGNAL, PROBABILITY, M1/M5/M15 & WAIT REASON) // ============================================================================ var table hud = table.new(position.top_right, 2, 8, bgcolor = color.new(#111317, 10), border_color = #2d313a, border_width = 1, frame_color = #3b4049, frame_width = 1) if showDashboard and barstate.islast statusTxt = canBuy or strategy.position_size > 0 ? (strategy.position_size > 0 ? "ACTIVE BUY" : "BUY NOW") : canSell or strategy.position_size < 0 ? (strategy.position_size < 0 ? "ACTIVE SELL" : "SELL NOW") : "WAIT" statusBg = str.contains(statusTxt, "BUY") ? color.new(#00c896, 20) : str.contains(statusTxt, "SELL") ? color.new(#ff4d6d, 20) : color.new(#b88a28, 25) // Preview projected SL/TP even while waiting projRisk = atr * slAtrMult projDir = nextDir == "BUY" ? 1 : -1 dispEnt = strategy.position_size != 0 ? trEntry : close dispSL = strategy.position_size != 0 ? trSL : close - projDir * projRisk dispTP1 = strategy.position_size != 0 ? trTP1 : close + projDir * projRisk * tp1RR dispTP2 = strategy.position_size != 0 ? trTP2 : close + projDir * projRisk * tp2RR table.cell(hud, 0, 0, "XAUUSD SCALP (" + speedMode + ")", text_color = #eaca82, text_size = size.small, bgcolor = #171a20, text_halign = text.align_left) table.cell(hud, 1, 0, statusTxt, text_color = color.white, text_size = size.small, bgcolor = statusBg, text_halign = text.align_right) table.cell(hud, 0, 1, "Probability", text_color = #a9afbc, text_size = size.tiny, text_halign = text.align_left) table.cell(hud, 1, 1, "BUY " + str.tostring(buyProb) + "% | SELL " + str.tostring(sellProb) + "%", text_color = buyProb >= 50 ? #67d6b2 : #f39098, text_size = size.tiny, text_halign = text.align_right) table.cell(hud, 0, 2, "Market Clarity", text_color = #a9afbc, text_size = size.tiny, text_halign = text.align_left) table.cell(hud, 1, 2, statusTxt == "WAIT" ? waitReason : "CLEAR (" + nextDir + " " + str.tostring(topProb) + "%)", text_color = statusTxt == "WAIT" ? #eaca82 : #67d6b2, text_size = size.tiny, text_halign = text.align_right) table.cell(hud, 0, 3, "MTF Structure", text_color = #a9afbc, text_size = size.tiny, text_halign = text.align_left) table.cell(hud, 1, 3, "Chart: " + m1Bias + " | M5: " + m5Bias + " | M15: " + m15Bias, text_color = color.white, text_size = size.tiny, text_halign = text.align_right) table.cell(hud, 0, 4, strategy.position_size != 0 ? "Stop (Break-Even)" : "Stop Loss (SL)", text_color = #a9afbc, text_size = size.tiny, text_halign = text.align_left) table.cell(hud, 1, 5, str.tostring(dispSL, format.mintick), text_color = #ff4d6d, text_size = size.tiny, text_halign = text.align_right) table.cell(hud, 0, 6, "Collect Profit (TP1)", text_color = #a9afbc, text_size = size.tiny, text_halign = text.align_left) table.cell(hud, 1, 6, str.tostring(dispTP1, format.mintick) + " (" + str.tostring(tp1ClosePct) + "%)", text_color = #00c896, text_size = size.tiny, text_halign = text.align_right) table.cell(hud, 0, 7, "Runner Target (TP2)", text_color = #a9afbc, text_size = size.tiny, text_halign = text.align_left) table.cell(hud, 1, 7, str.tostring(dispTP2, format.mintick), text_color = #38bdf8, text_size = size.tiny, text_halign = text.align_right)
